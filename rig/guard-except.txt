# --- an attribute on a definition nothing outside the file could see ------
#
# list_sort.c carried __attribute__((nonnull(2,3))) on the DEFINITION. It was
# never on the declaration in include/linux/list_sort.h, so no caller ever
# saw it and it constrained nothing outside that one translation unit; it let
# the compiler assume non-null while compiling list_sort() itself.
#
# It is a real loss and it is recorded as one rather than papered over: the
# Rust takes raw pointers and asserts nothing about them. Moving it to the
# header would make it apply to callers for the first time, which is a
# behaviour change upstream did not make, so it is deliberately NOT done here.
# The `likely(bits)` hit on the same body is a branch hint, reproduced as
# `if bits != 0` at list_sort_rs.rs:217.

list_sort            nonnull was on the definition only, visible to no caller; likely(bits) is reproduced at list_sort_rs.rs:217

# --- A CONDITIONAL THAT WAS DELIBERATELY NOT REPRODUCED --------------------
#
# THIS IS NOT A FALSE POSITIVE. Pass 1 is right: the four siphash buffer
# entry points each carry
#
#     #if defined(CONFIG_DCACHE_WORD_ACCESS) && BITS_PER_LONG == 64
#
# and lib/siphash_rs.rs compiles only the #else arm, on every configuration.
# The #if arm reads the final partial word with load_unaligned_zeropad(): one
# word load that may run past the end of the buffer, made safe by an
# __ex_table entry that catches the fault. Reproducing it from Rust means
# emitting kernel exception-table entries from asm!.
#
# THE HASH IS IDENTICAL either way -- both arms compute the same `b`, and
# upstream's own lib/tests/siphash_kunit.c passes against the ported build.
# What is lost is one load instead of up to three on the tail. CLAUDE.md:
# performance is recorded, never a veto.
#
# Recorded in full in logs/deviations.md, with the fix shape. These four lines
# are the acknowledgement that the check saw it and a human decided, which is
# the whole point of this file.

__siphash_unaligned  CONFIG_DCACHE_WORD_ACCESS arm deliberately not reproduced; portable tail always. Same hash, slower. logs/deviations.md
__siphash_aligned    as __siphash_unaligned; and this symbol is not built on x86-64 at all
__hsiphash_unaligned as __siphash_unaligned
__hsiphash_aligned   as __siphash_unaligned; and this symbol is not built on x86-64 at all

# --- mpicoder.c: BYTES_PER_MPI_LIMB #if/#elif is architecture selection,
# --- not a runtime guard -----------------------------------------------
#
# mpi_read_buffer and mpi_write_to_sgl each carry two
# `#if BYTES_PER_MPI_LIMB == 4 / #elif == 8 / #else #error` blocks in the
# pristine C, choosing __be32 vs __be64 and cpu_to_be32() vs cpu_to_be64().
# BYTES_PER_MPI_LIMB is 8 unconditionally on this architecture (mpi_limb_t
# is `unsigned long`, checked: lib/crypto/mpi/mpi-internal.h), so only the
# `== 8` arm ever compiles here -- the same class as mpih-div.c's
# UDIV_TIME/UMUL_TIME dead branch (this project's own module doc on that
# file). mpicoder_rs.rs's `alimb.to_be_bytes()` on a `c_ulong` (always
# 64-bit) IS that one surviving arm; there is no second arm to lose on a
# single-architecture port. Pass 1 flags `#if`/`#elif` lines as guards by
# construct, the same way it does for the CONFIG_DCACHE_WORD_ACCESS entries
# above, and cannot tell a live fork from a compile-time-constant one.

mpi_read_buffer   BYTES_PER_MPI_LIMB #if 4/#elif 8 is architecture selection (always 8 here, mpi_limb_t is unsigned long), not a runtime guard; mpicoder_rs.rs's to_be_bytes() on c_ulong is the one surviving arm
mpi_write_to_sgl  as mpi_read_buffer; same two BYTES_PER_MPI_LIMB #if/#elif blocks, same reasoning

kstrdup_quotable  WARN_ON(string_escape_mem(...) != dlen) reproduced as kernel::kabi_warn! in string_helpers_rs.rs kstrdup_quotable(), same condition and same file/line attribution (the whole point of the macro); same pattern as errseq_set/timerqueue_add above
